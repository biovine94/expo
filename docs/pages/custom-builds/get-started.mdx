// 2. AUTHENTICATION SCREEN
// ==========================================
function AuthScreen({ onLogin }) {
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [isSigningUp, setIsSigningUp] = useState(false);

  const handleAuth = () => {
    if (!email || !password) {
      Alert.alert('Error', 'Please enter both email and password.');
      return;
    }
    // Prototype login state trigger
    onLogin({ uid: 'user_123', email });
  };

  return (
    <View style={styles.authContainer}>
      <Text style={styles.logo}>🏎️ CarSeeker</Text>
      <Text style={styles.authHeader}>{isSigningUp ? 'Create an Account' : 'Welcome Back'}</Text>

      <TextInput
        style={styles.input}
        placeholder="Email"
        placeholderTextColor="#777"
        value={email}
        onChangeText={setEmail}
        autoCapitalize="none"
      />

      <TextInput
        style={styles.input}
        placeholder="Password"
        placeholderTextColor="#777"
        value={password}
        onChangeText={setPassword}
        secureTextEntry
      />

      <TouchableOpacity style={styles.btnPrimary} onPress={handleAuth}>
        <Text style={styles.btnPrimaryText}>{isSigningUp ? 'Sign Up' : 'Log In'}</Text>
      </TouchableOpacity>

      <TouchableOpacity onPress={() => setIsSigningUp(!isSigningUp)}>
        <Text style={styles.switchText}>
          {isSigningUp ? 'Already have an account? Log In' : "Don't have an account? Sign Up"}
        </Text>
      </TouchableOpacity>
    </View>
  );
}

// 3. SCANNER SCREEN (SEEK-STYLE AI SCAN)
// ==========================================
function ScannerScreen() {
  const [permission, requestPermission] = useCameraPermissions();
  const [lastSpot, setLastSpot] = useState(null);
  const [badges, setBadges] = useState([
    { id: '1', name: 'First Spot', unlocked: false },
    { id: '2', name: 'Mopar Hunter', unlocked: false },
    { id: '3', name: 'Pony Club', unlocked: false }
  ]);

  if (!permission) return <View style={styles.container} />;
  if (!permission.granted) {
    return (
      <View style={[styles.container, styles.centered]}>
        <Text style={styles.textLight}>Camera access needed for scanning cars.</Text>
        <TouchableOpacity style={[styles.btnPrimary, { marginTop: 15 }]} onPress={requestPermission}>
          <Text style={styles.btnPrimaryText}>Grant Permission</Text>
        </TouchableOpacity>
      </View>
    );
  }

  const simulateScan = () => {
    const mockCars = [
      { make: 'Plymouth', model: 'Hemi Cuda', year: '1970', isMopar: true },
      { make: 'Ford', model: 'Mustang GT', year: '2020', isMustang: true },
      { make: 'Dodge', model: 'Challenger Hellcat', year: '2022', isMopar: true }
    ];
    const detected = mockCars[Math.floor(Math.random() * mockCars.length)];
    setLastSpot(detected);
    checkBadges(detected);
  };

  const checkBadges = (car) => {
    let newlyEarned = [];
    setBadges(prev => prev.map(b => {
      if (b.unlocked) return b;
      let unlock = false;
      if (b.id === '1') unlock = true;
      if (b.id === '2' && car.isMopar) unlock = true;
      if (b.id === '3' && car.isMustang) unlock = true;

      if (unlock) {
        newlyEarned.push(b.name);
        return { ...b, unlocked: true };
      }
      return b;
    }));

    if (newlyEarned.length > 0) {
      Alert.alert('🏆 Badge Earned!', newlyEarned.join(', '));
    }
  };

  return (
    <View style={styles.container}>
      <View style={styles.cameraBox}>
        <CameraView style={{ flex: 1 }}>
          <View style={styles.targetFrame}>
            <Text style={styles.targetText}>ALIGN CAR IN FRAME</Text>
          </View>
        </CameraView>
        <TouchableOpacity style={styles.scanBtn} onPress={simulateScan}>
          <Text style={styles.scanBtnText}>SCAN CAR</Text>
        </TouchableOpacity>
      </View>

      <ScrollView style={{ flex: 1, padding: 15 }}>
        {lastSpot && (
          <View style={styles.cardGreenBorder}>
            <Text style={styles.textGreenSmall}>MATCH DETECTED</Text>
            <Text style={styles.cardTitle}>{lastSpot.year} {lastSpot.make} {lastSpot.model}</Text>
          </View>
        )}

        <Text style={styles.sectionHeader}>Badges Earned</Text>
        <View style={styles.rowBetween}>
          {badges.map(b => (
            <View key={b.id} style={[styles.badgeCard, b.unlocked ? styles.badgeUnlocked : styles.badgeLocked]}>
              <Text style={styles.badgeName}>{b.name}</Text>
              <Text style={styles.badgeStatus}>{b.unlocked ? '✅ Earned' : '🔒 Locked'}</Text>
            </View>
          ))}
        </View>
      </ScrollView>
    </View>
  );
}

// 6. GARAGE & PRIVATE BUILD LOG SCREEN
// ==========================================
function GarageScreen() {
  const [isPrivate, setIsPrivate] = useState(true);
  const [logs, setLogs] = useState([
    { id: '1', mod: 'Corsa Exhaust', cost: 1450 },
    { id: '2', mod: 'Dyno ECU Tune', cost: 800 }
  ]);
  const [modName, setModName] = useState('');
  const [modCost, setModCost] = useState('');

  const addMod = () => {
    if (!modName) return;
    setLogs([...logs, { id: Date.now().toString(), mod: modName, cost: parseFloat(modCost) || 0 }]);
    setModName('');
    setModCost('');
  };

  const totalCost = logs.reduce((sum, item) => sum + item.cost, 0);

  return (
    <ScrollView style={styles.container} contentContainerStyle={{ padding: 15 }}>
      <View style={styles.card}>
        <Text style={styles.cardTitle}>Build Privacy Status</Text>
        <Text style={styles.textSub}>{isPrivate ? '🔒 Private Garage (Only You)' : '🌐 Public Build'}</Text>
        <Switch 
          value={isPrivate} 
          onValueChange={setIsPrivate} 
          trackColor={{ true: '#FF4444', false: '#00FF66' }} 
        />
      </View>

      <View style={styles.cardGreenBorder}>
        <Text style={styles.textGreenSmall}>TOTAL BUILD INVESTMENT</Text>
        <Text style={styles.costBig}>${totalCost.toLocaleString()}</Text>
      </View>

      <View style={styles.card}>
        <Text style={styles.cardTitle}>+ Log Modification</Text>
        <TextInput 
          style={styles.input} 
          placeholder="Part Name" 
          placeholderTextColor="#777" 
          value={modName} 
          onChangeText={setModName} 
        />
        <TextInput 
          style={styles.input} 
          placeholder="Cost ($)" 
          placeholderTextColor="#777" 
          keyboardType="numeric" 
          value={modCost} 
          onChangeText={setModCost} 
        />
        <TouchableOpacity style={styles.btnPrimary} onPress={addMod}>
          <Text style={styles.btnPrimaryText}>Save Entry</Text>
        </TouchableOpacity>
      </View>

      <Text style={styles.sectionHeader}>Build History</Text>
      {logs.map(item => (
        <View key={item.id} style={[styles.card, styles.rowBetween, { marginBottom: 8 }]}>
          <Text style={styles.textLight}>{item.mod}</Text>
          <Text style={styles.textGreenBold}>${item.cost}</Text>
        </View>
      ))}
    </ScrollView>
  );
}

const Tab = createBottomTabNavigator();

export default function App() {
  const [currentUser, setCurrentUser] = useState(null);

  if (!currentUser) {
    return <AuthScreen onLogin={setCurrentUser} />;
  }

  return (
    <NavigationContainer>
      <Tab.Navigator
        screenOptions={{
          headerStyle: { backgroundColor: '#1E1E1E' },
          headerTintColor: '#00FF66',
          tabBarStyle: { backgroundColor: '#121212', borderTopColor: '#222' },
          tabBarActiveTintColor: '#00FF66',
          tabBarInactiveTintColor: '#888',
        }}
      >
        <Tab.Screen name="Scan" component={ScannerScreen} options={{ tabBarIcon: () => <Text>📷</Text> }} />
        <Tab.Screen name="Feed" component={FeedScreen} options={{ tabBarIcon: () => <Text>🔥</Text> }} />
        <Tab.Screen name="Groups" component={GroupsScreen} options={{ tabBarIcon: () => <Text>👥</Text> }} />
        <Tab.Screen name="Garage" component={GarageScreen} options={{ tabBarIcon: () => <Text>🏎️</Text> }} />
      </Tab.Navigator>
    </NavigationContainer>
  );
}
